33 constexpr uint32_t DimColor(const uint32_t COLOR, const float FACTOR) {
34 const auto R = static_cast<uint32_t>(static_cast<float>((COLOR >> 16U) & 0xFFU) * FACTOR);
35 const auto G = static_cast<uint32_t>(static_cast<float>((COLOR >> 8U) & 0xFFU) * FACTOR);
36 const auto B = static_cast<uint32_t>(static_cast<float>(COLOR & 0xFFU) * FACTOR);
37 return 0xFF000000U | (R << 16U) | (G << 8U) | B;
38 }

127 void Simulate(const float DELTA_TIME) {
128 for (OrbitingBody& body : m_Bodies) {
129 for (size_t age = TRAIL_LENGTH - 1; age > 0; --age) {
130 body.trailX.at(age) = body.trailX.at(age - 1);
131 body.trailY.at(age) = body.trailY.at(age - 1);
132 }
133 body.trailX.at(0) = CENTER_X + body.radius * std::cos(body.angle);
134 body.trailY.at(0) = CENTER_Y + body.radius * std::sin(body.angle);
135
136 body.angle = std::fmod(body.angle + body.angularSpeed * DELTA_TIME, 2.0F * std::numbers::pi_v<float>);
137 }
138 }

140 void BuildRenderPacket() {
141 m_Drawables.clear();

143 for (const auto& [OFFSET_X, OFFSET_Y] : SUN_OFFSETS) {
144 m_Drawables.push_back(liara_render_drawable_t {
145 .x = CENTER_X + OFFSET_X,

147 .color = SUN_COLOR,

149 }
150
151 for (const OrbitingBody& body : m_Bodies) {
152 for (size_t age = TRAIL_LENGTH; age > 0; --age) {
153 const size_t INDEX = age - 1;
154 m_Drawables.push_back(liara_render_drawable_t {
155 .x = body.trailX.at(INDEX),
156 .y = body.trailY.at(INDEX),
157 .color = DimColor(body.color, TRAIL_DIM_FACTORS.at(INDEX)),
158 });
159 }
160 m_Drawables.push_back(liara_render_drawable_t {
161 .x = CENTER_X + body.radius * std::cos(body.angle),
162 .y = CENTER_Y + body.radius * std::sin(body.angle),
163 .color = body.color,
164 });
165 }
166

170 m_Packet.background_color = BACKGROUND_COLOR;
