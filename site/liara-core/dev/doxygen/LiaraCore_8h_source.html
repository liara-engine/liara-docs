33 constexpr uint32_t DimColor(const uint32_t COLOR, const float FACTOR) {
34 const auto R = static_cast<uint32_t>(static_cast<float>((COLOR >> 16U) & 0xFFU) * FACTOR);
35 const auto G = static_cast<uint32_t>(static_cast<float>((COLOR >> 8U) & 0xFFU) * FACTOR);
36 const auto B = static_cast<uint32_t>(static_cast<float>(COLOR & 0xFFU) * FACTOR);
37 return 0xFF000000U | (R << 16U) | (G << 8U) | B;
38 }

126 void Simulate(const float DELTA_TIME) {
127 for (OrbitingBody& body : m_Bodies) {
128 for (size_t age = TRAIL_LENGTH - 1; age > 0; --age) {
129 body.trailX.at(age) = body.trailX.at(age - 1);
130 body.trailY.at(age) = body.trailY.at(age - 1);
131 }
132 body.trailX.at(0) = CENTER_X + body.radius * std::cos(body.angle);
133 body.trailY.at(0) = CENTER_Y + body.radius * std::sin(body.angle);
134
135 body.angle = std::fmod(body.angle + body.angularSpeed * DELTA_TIME, 2.0F * std::numbers::pi_v<float>);
136 }
137 }

139 void BuildRenderPacket() {
140 m_Drawables.clear();
141
142 for (const auto& [OFFSET_X, OFFSET_Y] : SUN_OFFSETS) {
143 m_Drawables.push_back(liara_render_drawable_t {
144 .x = CENTER_X + OFFSET_X,
145 .y = CENTER_Y + OFFSET_Y,
146 .color = SUN_COLOR,
147 });
148 }
149
150 for (const OrbitingBody& body : m_Bodies) {
151 for (size_t age = TRAIL_LENGTH; age > 0; --age) {
152 const size_t INDEX = age - 1;
153 m_Drawables.push_back(liara_render_drawable_t {
154 .x = body.trailX.at(INDEX),
155 .y = body.trailY.at(INDEX),
156 .color = DimColor(body.color, TRAIL_DIM_FACTORS.at(INDEX)),
157 });
158 }
159 m_Drawables.push_back(liara_render_drawable_t {
160 .x = CENTER_X + body.radius * std::cos(body.angle),
161 .y = CENTER_Y + body.radius * std::sin(body.angle),
162 .color = body.color,
163 });
164 }
165
166 m_Packet.struct_version = LIARA_RENDER_PACKET_VERSION;
167 m_Packet.grid_width = GRID_WIDTH;
168 m_Packet.grid_height = GRID_HEIGHT;
169 m_Packet.background_color = BACKGROUND_COLOR;
170 m_Packet.drawables = m_Drawables.data();
171 m_Packet.drawable_count = m_Drawables.size();
172 }
