33 constexpr uint32_t DimColor(const uint32_t color, const float factor) {
34 const auto r = static_cast<uint32_t>(static_cast<float>((color >> 16U) & 0xFFU) * factor);
35 const auto g = static_cast<uint32_t>(static_cast<float>((color >> 8U) & 0xFFU) * factor);
36 const auto b = static_cast<uint32_t>(static_cast<float>(color & 0xFFU) * factor);
37 return 0xFF000000U | (r << 16U) | (g << 8U) | b;
38 }

130 void Simulate(const float deltaTime) {
131 for (OrbitingBody& body : m_Bodies) {
132 for (size_t age = TRAIL_LENGTH - 1; age > 0; --age) {
133 body.m_TrailX.at(age) = body.m_TrailX.at(age - 1);
134 body.m_TrailY.at(age) = body.m_TrailY.at(age - 1);
135 }
136 body.m_TrailX.at(0) = CENTER_X + (body.m_Radius * std::cos(body.m_Angle));
137 body.m_TrailY.at(0) = CENTER_Y + (body.m_Radius * std::sin(body.m_Angle));
138
139 body.m_Angle =
140 std::fmod(body.m_Angle + (body.m_AngularSpeed * deltaTime), 2.0F * std::numbers::pi_v<float>);
141 }
142 }

144 void BuildRenderPacket() {
145 m_Drawables.clear();
146
147 for (const auto& [OFFSET_X, OFFSET_Y] : SUN_OFFSETS) {
148 m_Drawables.push_back(liara_render_drawable_t {
149 .x = CENTER_X + OFFSET_X,
150 .y = CENTER_Y + OFFSET_Y,
151 .color = SUN_COLOR,
152 });
153 }
154
155 for (const OrbitingBody& body : m_Bodies) {
156 for (size_t age = TRAIL_LENGTH; age > 0; --age) {
157 const size_t index = age - 1;
158 m_Drawables.push_back(liara_render_drawable_t {
159 .x = body.m_TrailX.at(index),
160 .y = body.m_TrailY.at(index),
161 .color = DimColor(body.m_Color, TRAIL_DIM_FACTORS.at(index)),
162 });
163 }
164 m_Drawables.push_back(liara_render_drawable_t {
165 .x = CENTER_X + (body.m_Radius * std::cos(body.m_Angle)),
166 .y = CENTER_Y + (body.m_Radius * std::sin(body.m_Angle)),
167 .color = body.m_Color,
168 });
169 }
170
171 m_Packet.struct_version = LIARA_RENDER_PACKET_VERSION;
172 m_Packet.grid_width = GRID_WIDTH;
173 m_Packet.grid_height = GRID_HEIGHT;
174 m_Packet.background_color = BACKGROUND_COLOR;
175 m_Packet.drawables = m_Drawables.data();
