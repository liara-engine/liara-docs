88 const uint8_t PIXEL_X = m_PixelPosX.at(i);

98 if (m_PixelPosY.at(i) >= m_ScreenHeight || m_PixelPosY.at(i) < 0) {
99 m_PixelVelocityY.at(i) = -m_PixelVelocityY.at(i);
100 m_PixelPosY.at(i) += m_PixelVelocityY.at(i);
101 }
102
103 m_Screen.at(m_PixelPosY.at(i)).at(m_PixelPosX.at(i)) = m_PixelColors.at(i);
104 m_Screen.at(PIXEL_Y).at(PIXEL_X) = 0xFF000000;
