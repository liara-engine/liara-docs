101static inline uint32_t liara_make_version(const uint32_t major, const uint32_t minor, const uint32_t patch) {
102 if (major > LIARA_VERSION_MAJOR_MASK) { return 0; }
103 if (minor > LIARA_VERSION_MINOR_MASK) { return 0; }
104 if (patch > LIARA_VERSION_PATCH_MASK) { return 0; }
105 return LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);

129static inline bool liara_try_make_version(const uint32_t major, const uint32_t minor, const uint32_t patch, uint32_t* out_version) {
130 if (out_version == NULL) { return false; }
131 if (major > LIARA_VERSION_MAJOR_MASK) { return false; }
132 if (minor > LIARA_VERSION_MINOR_MASK) { return false; }
133 if (patch > LIARA_VERSION_PATCH_MASK) { return false; }
134 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);

264static inline bool liara_version_lt(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
265 if (ignore_patch) {
266 if (LIARA_VERSION_MAJOR(v1) < LIARA_VERSION_MAJOR(v2)) return true;
267 if (LIARA_VERSION_MAJOR(v1) > LIARA_VERSION_MAJOR(v2)) return false;
268 return LIARA_VERSION_MINOR(v1) < LIARA_VERSION_MINOR(v2);
269 }
270 return v1 < v2;
271}
