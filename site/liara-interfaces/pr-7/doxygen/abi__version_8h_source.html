166static inline liara_version_compat_t liara_version_provides(const uint32_t provided, const uint32_t required) {
167 if (provided == required) { return LIARA_VERSION_COMPAT_EXACT; }
168 if (LIARA_VERSION_MAJOR(provided) != LIARA_VERSION_MAJOR(required)) { return LIARA_VERSION_COMPAT_INCOMPATIBLE; }
169 if (LIARA_VERSION_MAJOR(provided) == 0 && LIARA_VERSION_MINOR(provided) == 0) {
170 return provided == required ? LIARA_VERSION_COMPAT_EXACT : LIARA_VERSION_COMPAT_INCOMPATIBLE;
171 }
172 if (LIARA_VERSION_MINOR(provided) < LIARA_VERSION_MINOR(required)) { return LIARA_VERSION_COMPAT_DEGRADED; }
173 return LIARA_VERSION_COMPAT_COMPATIBLE;
174}
