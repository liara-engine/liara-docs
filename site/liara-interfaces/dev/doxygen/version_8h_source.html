74static inline bool liara_try_make_version(const uint32_t major, const uint32_t minor, const uint32_t patch, uint32_t* out_version) {
75 if (out_version == NULL) { return false; }
76 if (major > LIARA_VERSION_MAJOR_MASK) { return false; }
77 if (minor > LIARA_VERSION_MINOR_MASK) { return false; }
78 if (patch > LIARA_VERSION_PATCH_MASK) { return false; }
79 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);
80 return true;
81}

209static inline bool liara_version_lt(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
210 if (ignore_patch) {
211 if (LIARA_VERSION_MAJOR(v1) < LIARA_VERSION_MAJOR(v2)) return true;
212 if (LIARA_VERSION_MAJOR(v1) > LIARA_VERSION_MAJOR(v2)) return false;
213 return LIARA_VERSION_MINOR(v1) < LIARA_VERSION_MINOR(v2);
214 }
215 return v1 < v2;
