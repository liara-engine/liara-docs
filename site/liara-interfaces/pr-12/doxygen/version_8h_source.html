117static inline liara_result_t liara_try_make_version(const uint32_t major,
118 const uint32_t minor,
119 const uint32_t patch,
120 uint32_t* out_version) {
121 if (out_version == LIARA_NULL) { return LIARA_RESULT_NULL_POINTER; }
122 if (major > LIARA_VERSION_MAJOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
123 if (minor > LIARA_VERSION_MINOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
124 if (patch > LIARA_VERSION_PATCH_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
125 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);
126 return LIARA_RESULT_SUCCESS;
127}

229static inline int8_t liara_version_compare(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
230 uint32_t a = v1;
231 uint32_t b = v2;
232
233 // If ignoring patch, shift both versions right by the number of bits allocated for the patch version
234 if (ignore_patch) {
235 a >>= LIARA_VERSION_PATCH_BITS;
236 b >>= LIARA_VERSION_PATCH_BITS;
237 }
238
239 return LIARA_STATIC_CAST(int8_t, (LIARA_STATIC_CAST(int8_t, a > b) - LIARA_STATIC_CAST(int8_t, a < b)));
240}
