122static inline liara_result liara_try_make_version(const uint32_t major,
123 const uint32_t minor,
124 const uint32_t patch,
125 uint32_t* out_version) {
126 if (out_version == LIARA_NULL) { return LIARA_RESULT_NULL_POINTER; }
127 if (major > LIARA_VERSION_MAJOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
128 if (minor > LIARA_VERSION_MINOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
129 if (patch > LIARA_VERSION_PATCH_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
130 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);
131 return LIARA_RESULT_SUCCESS;
132}

380static inline int8_t liara_version_compare(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
381 uint32_t a = v1;
382 uint32_t b = v2;
383
384 // If ignoring patch, shift both versions right by the number of bits allocated for the patch version
385 if (ignore_patch) {
386 a >>= LIARA_VERSION_PATCH_BITS;
387 b >>= LIARA_VERSION_PATCH_BITS;
388 }
389
390 return LIARA_STATIC_CAST(int8_t, (LIARA_STATIC_CAST(int8_t, a > b) - LIARA_STATIC_CAST(int8_t, a < b)));
391}
