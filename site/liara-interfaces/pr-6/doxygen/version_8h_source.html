77static inline bool liara_try_make_version(const uint32_t major,
78 const uint32_t minor,
79 const uint32_t patch,
80 uint32_t* out_version) {
81 if (out_version == NULL) { return false; }
82 if (major > LIARA_VERSION_MAJOR_MASK) { return false; }
83 if (minor > LIARA_VERSION_MINOR_MASK) { return false; }
84 if (patch > LIARA_VERSION_PATCH_MASK) { return false; }
85 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);

220static inline bool liara_version_lt(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
221 if (ignore_patch) {
222 if (LIARA_VERSION_MAJOR(v1) < LIARA_VERSION_MAJOR(v2)) return true;
223 if (LIARA_VERSION_MAJOR(v1) > LIARA_VERSION_MAJOR(v2)) return false;
224 return LIARA_VERSION_MINOR(v1) < LIARA_VERSION_MINOR(v2);
225 }
226 return v1 < v2;
227}
