112static inline liara_result liara_try_make_version(const uint32_t major,
113 const uint32_t minor,
114 const uint32_t patch,
115 uint32_t* out_version) {
116 if (out_version == NULL) { return LIARA_RESULT_NULL_POINTER; }
117 if (major > LIARA_VERSION_MAJOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
118 if (minor > LIARA_VERSION_MINOR_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
119 if (patch > LIARA_VERSION_PATCH_MASK) { return LIARA_RESULT_OUT_OF_RANGE; }
120 *out_version = LIARA_MAKE_VERSION_UNSAFE(major, minor, patch);
121 return LIARA_RESULT_SUCCESS;
122}

215static inline bool liara_version_eq(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
216 if (ignore_patch) {

255static inline bool liara_version_lt(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
256 if (ignore_patch) {
257 if (LIARA_VERSION_MAJOR(v1) < LIARA_VERSION_MAJOR(v2)) { return true; }
258 if (LIARA_VERSION_MAJOR(v1) > LIARA_VERSION_MAJOR(v2)) { return false; }
259 return LIARA_VERSION_MINOR(v1) < LIARA_VERSION_MINOR(v2);
260 }
261 return v1 < v2;
262}

280static inline bool liara_version_lte(const uint32_t v1, const uint32_t v2, const bool ignore_patch) {
